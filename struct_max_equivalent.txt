struct Entry {
    int key;
    char data[64];
};

struct Entry max_entry(struct Entry entries[], int count) {
    struct Entry best = entries[0];
    for (int i = 1; i < count; i++) {
        if (best.key < entries[i].key) {
            best = entries[i];
        }
    }
    return best;
}

struct Entry {
    int key;
    char data[64];
};

struct Entry max_entry(struct Entry entries[], int count) {
    struct Entry best = entries[0];
    for (int i = 1; i < count; i++) {
        best = (best.key < entries[i].key) ? entries[i] : best;
    }
    return best;
}
